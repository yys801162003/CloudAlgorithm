cmake_minimum_required(VERSION 3.16)
project(SmokeDetectorCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs dnn)
add_library(SmokeEngine SmokeEngine.cpp)
target_include_directories(SmokeEngine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(SmokeEngine PUBLIC ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(SmokeEngine PUBLIC /utf-8 /EHsc)
endif()
add_executable(SmokeTests tests/SmokeTests.cpp)
target_link_libraries(SmokeTests PRIVATE SmokeEngine)
enable_testing()
add_test(NAME SmokeEngineTests COMMAND SmokeTests ${CMAKE_CURRENT_SOURCE_DIR})
